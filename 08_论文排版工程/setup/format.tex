% setup/format.tex - 格式设定

% ========== xeCJK设置 ==========
\xeCJKsetup{silent}

% ========== 字体设置 (六步法标准) ==========
% 华文行楷（封面标题用）
\setCJKfamilyfont{xingkai}{STXingkai}
\newcommand{\xingkai}{\CJKfamily{xingkai}}

% ========== 字体设置 ==========
% 中文字体（xeCJK自动处理中文回退）
\setCJKmainfont{SimSun}[AutoFakeBold=2.5]
\setCJKsansfont{SimHei}[AutoFakeBold=2.5]
\setCJKmonofont{FangSong}
\setmainfont{Times New Roman}

% ========== 章节标题格式 (匹配Word模板) ==========
% 一级标题 (第1章, 第2章...)
% Word规范：小二号黑体居中，实际16pt
\ctexset{
    chapter = {
        format = \centering\bfseries\zihao{2}\heiti,
        name = {第,章},
        number = \chinese{chapter},
        aftername = \quad,
        beforeskip = 20pt,
        afterskip = 18pt,
        fixskip = true,
    },
    % 二级标题 (1.1, 2.1...)
    % Word规范：小四号黑体，实际14pt
    section = {
        format = \raggedright\bfseries\zihao{5}\heiti,
        aftername = \quad,
        beforeskip = 12pt,
        afterskip = 10pt,
        fixskip = true,
    },
    % 三级标题 (1.1.1...)
    % Word规范：小四号黑体，实际13pt
    subsection = {
        format = \raggedright\bfseries\zihao{-4}\heiti,
        aftername = \quad,
        beforeskip = 8pt,
        afterskip = 6pt,
        fixskip = true,
    },
    % 四级标题
    subsubsection = {
        format = \raggedright\bfseries\zihao{-4}\heiti,
        aftername = \quad,
        beforeskip = 6pt,
        afterskip = 6pt,
        fixskip = true,
    }
}
